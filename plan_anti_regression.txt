  2.1 - Remplacement Redux par Zustand

  npm install zustand

  // mobile/src/store/useAppStore.ts
  import { create } from 'zustand'
  import { devtools, persist } from 'zustand/middleware'

  interface AppState {
    // State
    categories: Category[]
    products: Product[]
    user: User | null

    // Actions
    setCategories: (categories: Category[]) => void
    setProducts: (products: Product[]) => void
    setUser: (user: User | null) => void

    // Async actions
    fetchData: () => Promise<void>
    logout: () => void
  }

  export const useAppStore = create<AppState>()(
    devtools(
      persist(
        (set, get) => ({
          // Initial state
          categories: [],
          products: [],
          user: null,

          // Actions simples et claires
          setCategories: (categories) => set({ categories }),
          setProducts: (products) => set({ products }),
          setUser: (user) => set({ user }),

          // Fetch tout en une fois
          fetchData: async () => {
            const [categoriesRes, productsRes] = await Promise.all([
              fetch('/api/categories'),
              fetch('/api/products')
            ])

            const categories = await categoriesRes.json()
            const products = await productsRes.json()

            set({
              categories: categories.data,
              products: products.data
            })
          },

          // Logout simple
          logout: () => {
            set({ user: null, products: [], categories: [] })
          }
        }),
        {
          name: 'app-storage',
          // Automatically sync with AsyncStorage
        }
      )
    )
  )

  Utilisation Ultra-Simple:
  // Dans n'importe quel composant
  const { categories, products, fetchData } = useAppStore()

  // C'est TOUT ! Pas de dispatch, pas de selectors, pas de Redux

  4.2 - Hot Reload Amélioré

  // mobile/src/utils/devTools.ts
  if (__DEV__) {
    // Clear cache automatiquement à chaque reload
    const clearCache = async () => {
      await AsyncStorage.clear()
      console.log('🧹 Cache cleared on reload')
    }

    // Vérifier l'intégrité des données
    const checkDataIntegrity = () => {
      const state = store.getState()
      if (!state.products.categories.length) {
        console.warn('⚠️ Categories vides - fetching...')
        store.dispatch(fetchCategories())
      }
    }

    // Exécuter à chaque hot reload
    module.hot?.accept(() => {
      clearCache()
      checkDataIntegrity()
    })
  }
